CLI 命令参考

构建命令

# 完整构建
bun run main.ts

# TypeScript 类型检查
bun --check main.ts

目录结构命令

# 查看生成的文件
ls -R dist/

# 查看内容目录
ls -R content/

# 查看主题目录
ls -R themes/

清理命令

# 清理构建输出
rm -rf dist/

# 重新构建
bun run main.ts
